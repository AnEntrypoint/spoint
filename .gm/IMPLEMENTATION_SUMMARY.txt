INTEGRATION EXAMPLE
===================

  import { defineQuestSystem } from './QuestSystem.js'
  import { defineStatsSystem } from './StatsSystem.js'

  export const server = {
    async setup(ctx) {
      // Initialize quest system
      ctx.quests = defineQuestSystem({
        quests: {
          'kill-rats': {
            title: 'Slay Rats',
            objectives: [{ type: 'killN', count: 5, enemyType: 'rat' }],
            rewards: { xp: 50, items: { copper: 2 } }
          }
        }
      }, ctx)

      // Initialize stats system
      ctx.stats = defineStatsSystem({
        baseStats: { health: 100, damage: 10, defense: 5 },
        statScaling: { health: 10, damage: 0.5, defense: 0.25 },
        equipment: { weapon: [...], armor: [...] }
      }, ctx)
    },
    
    onPlayerJoin(ctx, playerId) {
      ctx.quests.startQuest(playerId, 'kill-rats')
      ctx.stats.equipItem(playerId, 'iron-sword')
    }
  }
